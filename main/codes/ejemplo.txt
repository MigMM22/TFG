import qsimov as qj
qa = qj.QGate(2, 0, "CustGate2")
qa.add_operation("H", targets=1)
qa.add_operation("X", targets=[1])
qg = qj.QGate(3, 0, "CustGate")
qg.add_operation(qa, targets = [1,2])
qg.add_operation("sqrtSWAP", targets = [1,2], controls = 0)
qg.add_operation("H", targets=1)
qg.add_operation("X", targets=[1])
qc = qj.QCircuit(4, 3, "Teleportation", ancilla=(0, 0))
qc.add_operation(qg, targets=[1,2,3])
