import qsimov as qj

qg = qj.QGate(1, 1, "CustGate")
qg.add_operation("H", targets=[0])
qg.add_operation("X", targets=[0])

qc = qj.QCircuit(2, 2, "Name")

qc.add_operation(qg, targets = [0], c_targets = 0)
qc.add_operation(qg, targets = [1])
qc.add_operation("MEASURE", targets=(1,0), outputs=(0,1))
